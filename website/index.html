Quickstart

Start in 3 Lines of Code

Install the package and give your agent immediate Solana protocol access.

pnpm add solana-agent-kit @solana/web3.js
agent.ts 
import { SolanaAgentKit, createSolanaTools } from "solana-agent-kit";

// Initialize agent with private key and RPC
const agent = new SolanaAgentKit(
  process.env.SOLANA_PRIVATE_KEY!,
  "https://api.mainnet-beta.solana.com",
  process.env.OPENAI_API_KEY!
);

// Create LangChain or Vercel AI SDK tools
const tools = createSolanaTools(agent);

// Agent is now fully equipped to autonomously trade, stake & deploy!
console.log("Agent initialized with", tools.length, "Solana actions.");
